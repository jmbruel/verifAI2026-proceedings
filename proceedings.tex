% proceedings.tex
%
% Master LaTeX pdfpages assembly file for the VERIF-AI 2026 LNCS proceedings
% (Springer LNCS Volume 17411; VERIF-AI 2026, Villebrumier, France, 8-11 March 2026)
%
% -----------------------------------------------------------------------
% WHY pdfpages?
%   Springer typesets each paper independently from its own folder (see
%   springer-files/CS_Editors_Proceedings_guidelines.pdf, Sect. 5.6) and
%   does NOT want a single merged LaTeX source file. Attempting to \input
%   25 standalone author .tex files fails due to conflicting preambles,
%   packages, custom macros, and bibliographies.
%
%   Instead, this file is a local editorial assembly harness:
%     1. Compile each LaTeX paper independently in its own folder:
%          make paper DIR=papers/p1992      (or: make papers)
%        to verify compilability (updates Compilation Status in
%        tracking/paper-tracking.md) and produce its PDF.
%     2. Uncomment the paper's \includepaper line below (pointing to its
%        compiled or camera-ready PDF).
%     3. Run `make proceedings` to generate proceedings.pdf with LNCS
%        front matter, Table of Contents, and sequential pagination to
%        verify total page count and assign Starting Pages for
%        submission/toc.csv.
%
% FOLDER NAMING LIFECYCLE:
%   - Initial collection: papers/p<submission-id>/ (e.g., papers/p1992/paper.pdf)
%   - Assigned volume number: LNCS 17411. Once final paper order & pagination
%     are locked, rename each folder to 17411<4-digit-starting-page>
%     (e.g., papers/174110001/) and update the PDF path if re-running locally.
%
% TOPICAL SECTIONS (Part Titles):
%   Replace each PLACEHOLDER in \part{} with the agreed topical heading and
%   reorder \includepaper entries to match submission/toc.csv.
%   \setcounter{secnumdepth}{-2} suppresses "Part I / Part II" numbering
%   per Springer guidelines (no session numbering in ToC).
% -----------------------------------------------------------------------

\documentclass[runningheads]{llncs}

\usepackage[T1]{fontenc}
\usepackage{graphicx}
\usepackage{pdfpages}
\usepackage[hidelinks]{hyperref}

% Suppress "Part I / Part II" prefixes so topical parts match LNCS style
\setcounter{secnumdepth}{-2}

% Bookmark levels for llncs.cls's native \l@title and \l@author ToC macros
\makeatletter
\providecommand*{\toclevel@title}{0}
\providecommand*{\toclevel@author}{1}
\makeatother

% -----------------------------------------------------------------------
% Helper macro to include one paper's compiled PDF and register it in TOC
% Usage: \includepaper{<path/to/paper.pdf>}{<Paper Title>}{<Author List>}
%
% Note: pagecommand={\thispagestyle{plain}} stamps continuous volume page
% numbers in the bottom footer while leaving each paper's own top running
% head untouched. Switch to pagecommand={\thispagestyle{empty}} if you want
% only the embedded author PDFs without footer page numbers.
% -----------------------------------------------------------------------
\newcommand{\includepaper}[3]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{title}{#2}%
  \addcontentsline{toc}{author}{#3}%
  \includepdf[pages=-,pagecommand={\thispagestyle{plain}}]{#1}%
}
